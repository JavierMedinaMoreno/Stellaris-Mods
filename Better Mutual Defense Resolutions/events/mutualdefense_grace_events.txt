namespace = mutualdefense_grace

country_event = {
	id = mutualdefense_grace.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_country_type = default
		OR = {
			is_any_mutualdefense_resolution_active = yes
			has_country_flag = md_breach_grace_started
			has_country_flag = md_personality_swapped
		}
	}

	immediate = {
		if = {
			limit = {
				is_violating_mutualdefense_resolution = yes
			}
			if = {
				limit = {
					NOT = { has_country_flag = md_breach_grace_started }
				}
				set_timed_country_flag = { flag = md_breach_grace_period years = @GRACE_PERIOD_YEARS }
				set_country_flag = md_breach_grace_started
				add_modifier = { modifier = "mutual_defense_grace_period" years = @GRACE_PERIOD_YEARS }
				if = {
					limit = { is_ai = no }
					country_event = { id = mutualdefense_grace.2 }
				}
			}
			if = {
				limit = {
					is_ai = yes
					NOT = { has_country_flag = md_personality_swapped }
				}
				country_event = { id = mutualdefense_grace.10 }
			}
		}
		else = {
			if = {
				limit = { has_country_flag = md_breach_grace_started }
				remove_country_flag = md_breach_grace_started
				remove_country_flag = md_breach_grace_period
				remove_modifier = "mutual_defense_grace_period"
				if = {
					limit = {
						is_ai = no
						navy_is_below_active_mutualdefense_requirement = no
					}
					country_event = { id = mutualdefense_grace.3 }
				}
			}
			if = {
				limit = {
					has_country_flag = md_personality_swapped
				}
				country_event = { id = mutualdefense_grace.11 }
			}
		}
	}
}
