namespace = gateway_calc

# ==============================================================================
# 1. CORE RECALCULATION EVENT
# Scope: Country
# ==============================================================================
country_event = {
    id = gateway_calc.1
    is_triggered_only = yes
    hide_window = yes

    immediate = {
        calculate_gateway_capacity = yes
    }
}

# ==============================================================================
# 2. SAVE LOAD BROADCASTER
# Scope: None (fired from on_single_player_save_game_load)
# ==============================================================================
event = {
    id = gateway_calc.2
    is_triggered_only = yes
    hide_window = yes

    immediate = {
        every_country = {
            limit = {
                is_country_type = default
            }
            country_event = { id = gateway_calc.1 }
        }
    }
}

# ==============================================================================
# 3. STARBASE / OUTPOST / WAYSTATION DISPATCHER
# Scope: Starbase (outpost built/lost/transferred) or Solar System (waystations in neutral space)
# from = Owner / Builder / Previous Owner Country
# ==============================================================================
event = {
    id = gateway_calc.30
    is_triggered_only = yes
    hide_window = yes

    immediate = {
        # 1. Direct owner on this object (starbase owner or claimed solar system owner)
        if = {
            limit = { exists = owner }
            owner = {
                country_event = { id = gateway_calc.1 }
            }
        }
        # 2. Fallback: if this is a sub-object, scope into containing solar system owner
        else_if = {
            limit = {
                exists = solar_system
                solar_system = { exists = owner }
            }
            solar_system = {
                owner = {
                    country_event = { id = gateway_calc.1 }
                }
            }
        }
        # 3. Recalculate 'from' when distinct from 'owner' (or when owner does not exist):
        #    - on_waystation_built: neutral system has no owner; 'from' is nomad builder country
        #    - on_waystation_lost: starbase owner may be cleared; 'from' is previous owner country
        #    - on_starbase_transfer: 'owner' is new owner, 'from' is former owner (both recalculate)
        if = {
            limit = {
                exists = from
                NOT = {
                    exists = owner
                    owner = { is_same_value = from }
                }
            }
            from = {
                country_event = { id = gateway_calc.1 }
            }
        }
    }
}

# ==============================================================================
# 4. STARBASE DESTROYED DISPATCHER
# Scope: Starbase
# from = Fleet (Attacker) - Attacker does not recalculate, only victim starbase owner
# ==============================================================================
event = {
    id = gateway_calc.32
    is_triggered_only = yes
    hide_window = yes

    immediate = {
        # Recalculate capacity exclusively for the victim starbase owner
        if = {
            limit = { exists = owner }
            owner = {
                country_event = { id = gateway_calc.1 }
            }
        }
        else_if = {
            limit = {
                exists = solar_system
                solar_system = { exists = owner }
            }
            solar_system = {
                owner = {
                    country_event = { id = gateway_calc.1 }
                }
            }
        }
    }
}
