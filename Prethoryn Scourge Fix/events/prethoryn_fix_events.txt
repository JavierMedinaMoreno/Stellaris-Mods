# 10-day coordinator loop running on the Prethoryn Swarm country (root = swarm).
# Evaluates galaxy star systems to assign siege and capture task forces to contested colonies.
country_event = {
	id = prethoryn_fix.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		every_system = {
			limit = {
				# Swarm or loaned dummy fleet must be physically present in system
				any_fleet_in_system = {
					exists = owner
					owner = {
						OR = {
							is_same_value = root
							is_country_type = swarm_dummy
						}
					}
				}

				# System starbase must not be controlled by a hostile empire
				NOR = {
					AND = {
						exists = starbase
						starbase = { owner = { is_hostile = root } }
					}
				}

				# System contains an unconquered hostile colony or an anchored colony needing cleanup
				any_system_planet = {
					is_star = no
					OR = {
						AND = {
							is_colony = yes
							owner = { is_hostile = root }
							NOR = {
								owner = {
									OR = {
										is_country_type = swarm
										is_country_type = swarm_dummy
									}
								}
								AND = {
									exists = controller
									controller = {
										OR = {
											is_country_type = swarm
											is_country_type = swarm_dummy
										}
									}
								}
							}
						}
						has_planet_flag = prethoryn_has_siege_anchor
					}
				}
			}

			every_system_planet = {
				limit = {
					is_star = no
					OR = {
						AND = {
							is_colony = yes
							owner = { is_hostile = root }
							NOR = {
								owner = {
									OR = {
										is_country_type = swarm
										is_country_type = swarm_dummy
									}
								}
								AND = {
									exists = controller
									controller = {
										OR = {
											is_country_type = swarm
											is_country_type = swarm_dummy
										}
									}
								}
							}
						}
						has_planet_flag = prethoryn_has_siege_anchor
					}
				}
				planet_event = { id = prethoryn_fix.10 }
			}
		}

		country_event = { id = prethoryn_fix.1 days = 10 }
		country_event = { id = prethoryn_fix.60 }
	}
}

# Schedules 5-day delayed transport army splitting upon planetary conquest
country_event = {
	id = prethoryn_fix.54
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			is_country_type = swarm
			is_country_type = swarm_dummy
		}
	}

	immediate = {
		if = {
			limit = { is_country_type = swarm_dummy }
			random_country = {
				limit = {
					is_country_type = swarm

			}
		}
		else = {
			country_event = { id = prethoryn_fix.60 days = 5 }
		}
	}
}

# Transport Fleet Splitting: Splits merged armies exceeding 20 into standard task forces
country_event = {
	id = prethoryn_fix.60
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_country_type = swarm
		NOT = { is_country_type = swarm_dummy }
	}

	immediate = {
		owner_species = { save_event_target_as = prethoryn_species }
		every_owned_fleet = {
			limit = {
				is_ship_class = shipclass_transport
				NOT = { has_fleet_flag = prethoryn_assigned_transport_fleet }
				exists = solar_system
				count_controlled_ship = {
					limit = { is_ship_class = shipclass_transport }
					count > 20
				}
			}
			save_event_target_as = parent_transport_fleet
			solar_system = {
				save_event_target_as = transport_location_system
			}
			root = {
				create_fleet = {
					name = "NAME_Prethoryn_Armies"
					effect = {
						set_owner = root
						while = {
							count = 20
							create_army_transport = {
								ship_name = "NAME_Prethoryn_Transport"
								graphical_culture = "swarm_01"
								army_name = "NAME_Prethoryn_Invaders"
								army_type = "swarm_army"
								species = event_target:prethoryn_species
							}
						}
						set_location = {
							target = event_target:parent_transport_fleet
							distance = 5
							angle = random
						}
					}
				}
			}
			while = {
				count = 20
				random_controlled_ship = {
					limit = { is_ship_class = shipclass_transport }
					destroy_ship = this
				}
			}
		}
	}
}

# Startup Hook (on_country_created): Instantiates swarm_dummy and starts coordinator loop
country_event = {
	id = prethoryn_fix.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_country_type = swarm
		NOT = { is_country_type = swarm_dummy }
	}

	immediate = {
		set_country_flag = prethoryn_fix_loop_active

		if = {
			limit = {
				NOT = {
					any_country = { is_country_type = swarm_dummy }
				}
			}
			owner_species = { save_event_target_as = prethoryn_species }
			create_country = {
				name = "NAME_Prethoryn_Scourge"
				type = "swarm_dummy"
				species = event_target:prethoryn_species
				name_list = "Prethoryn"
				flag = {
					icon = { category = "special" file = "the_swarm.dds" }
					background = { category = "backgrounds" file = "new_dawn.dds" }
					colors = { "orange" "grey" "null" "null" }
				}
				effect = {
					save_global_event_target_as = prethoryn_dummy_country
				}
			}
		}

		every_country = {
			establish_communications_no_message = event_target:prethoryn_dummy_country
		}
		establish_communications_no_message = event_target:prethoryn_dummy_country
		set_faction_hostility = {
			target = event_target:prethoryn_dummy_country
			set_hostile = no
			set_neutral = no
			set_friendly = yes
		}
		event_target:prethoryn_dummy_country = {
			establish_communications_no_message = root
			set_faction_hostility = {
				target = root
				set_hostile = no
				set_neutral = no
				set_friendly = yes
			}
		}

		if = {
			limit = {
				has_global_flag = custom_difficulty_active
			}
			event_target:prethoryn_dummy_country = {
				set_country_flag = custom_difficulty_crisis_flag
				set_variable = { which = custom_difficulty_scaled_difficulty_bonuses value = 3.0 }
				country_event = { id = custom_difficulty.80 }
				country_event = { id = custom_difficulty.85 }
			}
		}

		country_event = { id = prethoryn_fix.1 }
	}
}

# Planet-scoped Colony Task Force Coordinator
# root: Target colony entity
# from: Prethoryn Swarm country entity
planet_event = {
	id = prethoryn_fix.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_star = no
		OR = {
			is_colony = yes
			has_planet_flag = prethoryn_has_siege_anchor
		}
	}

	immediate = {
		save_event_target_as = target_colony

		# Terminal Cleanup: Releases fleets when colony is conquered, purged, or infested
		if = {
			limit = {
				OR = {
					AND = {
						exists = owner
						owner = { is_country_type = swarm }
					}
					AND = {
						exists = controller
						controller = { is_country_type = swarm }
					}
					is_colony = no
					is_planet_class = pc_infested
				}
			}
			solar_system = {
				every_fleet_in_system = {
					limit = {
						OR = {
							has_fleet_flag = prethoryn_anchor@root
							has_fleet_flag = prethoryn_transport@root
						}
					}
					remove_fleet_flag = prethoryn_anchor@root
					remove_fleet_flag = prethoryn_transport@root
					remove_fleet_flag = prethoryn_assigned_combat_fleet
					remove_fleet_flag = prethoryn_assigned_transport_fleet
					remove_fleet_flag = prethoryn_loaned_combat_lock
					remove_fleet_flag = prethoryn_en_route@root
					clear_fleet_actions = this
					remove_auto_move_target = yes
					clear_orders = yes

					if = {
						limit = {
							owner = { is_country_type = swarm_dummy }
						}
						set_owner = from
					}
				}
			}
			from = {
				every_owned_fleet = {
					limit = {
						has_fleet_flag = prethoryn_transport@root
					}
					remove_fleet_flag = prethoryn_transport@root
					remove_fleet_flag = prethoryn_assigned_transport_fleet
					clear_fleet_actions = this
					remove_auto_move_target = yes
					clear_orders = yes
				}
			}
			every_country = {
				limit = { is_country_type = swarm_dummy }
				every_owned_fleet = {
					limit = {
						has_fleet_flag = prethoryn_anchor@root
					}
					remove_fleet_flag = prethoryn_anchor@root
					remove_fleet_flag = prethoryn_assigned_combat_fleet
					remove_fleet_flag = prethoryn_loaned_combat_lock
					remove_fleet_flag = prethoryn_en_route@root
					clear_fleet_actions = this
					remove_auto_move_target = yes
					clear_orders = yes
					set_owner = from
				}
			}
			remove_planet_flag = prethoryn_has_siege_anchor
		}
		else = {
			# System Boundary Safety Release: Strips anchor if warship exited the star system
			every_country = {
				limit = {
					OR = {
						is_country_type = swarm
						is_country_type = swarm_dummy
					}
				}
				every_owned_fleet = {
					limit = {
						has_fleet_flag = prethoryn_anchor@root
						NOT = { has_fleet_flag = prethoryn_en_route@root }
						NOT = {
							solar_system = {
								any_system_planet = { is_same_value = root }
							}
						}
					}
					remove_fleet_flag = prethoryn_anchor@root
					remove_fleet_flag = prethoryn_assigned_combat_fleet
					remove_fleet_flag = prethoryn_loaned_combat_lock
					remove_auto_move_target = yes
					if = {
						limit = {
							owner = { is_country_type = swarm_dummy }
						}
						set_owner = from
					}
				}
			}

			# Arrival Check: Remove transit flag upon entering the host system
			solar_system = {
				every_fleet_in_system = {
					limit = {
						has_fleet_flag = prethoryn_anchor@root
						has_fleet_flag = prethoryn_en_route@root
					}
					remove_fleet_flag = prethoryn_en_route@root
				}
			}

			# Section A: Combat Fleet Anchoring & Persistent Siege

			# Case 1: Colony has no anchor combat fleet assigned
			if = {
				limit = {
					solar_system = {
						NOT = {
							any_fleet_in_system = {
								has_fleet_flag = prethoryn_anchor@root
							}
						}
					}
				}
				remove_planet_flag = prethoryn_has_siege_anchor

				# Sub-tier 1A: Local military fleet within 40 units
				solar_system = {
					random_fleet_in_system = {
						limit = {
							owner = { is_country_type = swarm }
							is_ship_class = shipclass_military
							is_in_combat = no
							NOT = { has_fleet_flag = prethoryn_assigned_combat_fleet }
							distance = {
								source = root
								max_distance <= 40
								same_solar_system = yes
							}
						}
						if = {
							limit = {
								NOT = { exists = event_target:prethoryn_dummy_country }
							}
							random_country = {
								limit = { is_country_type = swarm_dummy }
								save_global_event_target_as = prethoryn_dummy_country
							}
						}
						set_owner = event_target:prethoryn_dummy_country
						set_fleet_bombardment_stance = armageddon
						set_fleet_stance = aggressive
						set_fleet_flag = prethoryn_anchor@root
						set_fleet_flag = prethoryn_assigned_combat_fleet
						clear_orders = yes
						clear_fleet_actions = this
						remove_auto_move_target = yes
						queue_actions = {
							move_to = root
							orbit_planet = root
							repeat = {
								while = {
									id = "prethoryn_hold_colony_siege_1a"
									root = {
										NOR = {
											AND = {
												exists = owner
												owner = {
													OR = {
														is_country_type = swarm
														is_country_type = swarm_dummy
													}
												}
											}
											AND = {
												exists = controller
												controller = {
													OR = {
														is_country_type = swarm
														is_country_type = swarm_dummy
													}
												}
											}
											is_colony = no
											is_planet_class = pc_infested
										}
									}
								}
								wait = 30
							}
						}
						root = { set_planet_flag = prethoryn_has_siege_anchor }
					}
				}
				# Sub-tier 1B: Any unassigned military fleet in host system
				if = {
					limit = { NOT = { has_planet_flag = prethoryn_has_siege_anchor } }
					solar_system = {
						random_fleet_in_system = {
							limit = {
								owner = { is_country_type = swarm }
								is_ship_class = shipclass_military
								is_in_combat = no
								NOT = { has_fleet_flag = prethoryn_assigned_combat_fleet }
							}
							if = {
								limit = {
									NOT = { exists = event_target:prethoryn_dummy_country }
								}
								random_country = {
									limit = { is_country_type = swarm_dummy }
									save_global_event_target_as = prethoryn_dummy_country
								}
							}
							set_owner = event_target:prethoryn_dummy_country

							clear_orders = yes
